package installer

import (
	"crypto/sha256"
	"encoding/hex"
	"os"
	"path/filepath"
	"strings"
	"testing"
)

func put(t *testing.T, root, rel, text string) {
	t.Helper()
	p := filepath.Join(root, rel)
	if err := os.MkdirAll(filepath.Dir(p), 0755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(p, []byte(text), 0644); err != nil {
		t.Fatal(err)
	}
}
func must(t *testing.T, err error) {
	t.Helper()
	if err != nil {
		t.Fatal(err)
	}
}
func exists(root, rel string) bool { _, err := os.Stat(filepath.Join(root, rel)); return err == nil }
func block() string                { return blockBegin + "\nmanaged\n" + blockEnd + "\n" }

func TestAudit_F01_UninstallPreservesNewConsumerProse(t *testing.T) {
	root := t.TempDir()
	put(t, root, agentsTarget, "# Agents\n\n"+block()+"\n## Consumer rule\nDo not deploy without approval.\n")
	removeAgentsBlock(root, &strings.Builder{})
	got, err := os.ReadFile(filepath.Join(root, agentsTarget))
	if err != nil || !strings.Contains(string(got), "Do not deploy without approval.") {
		t.Fatalf("consumer prose lost: exists=%v error=%v", exists(root, agentsTarget), err)
	}
}
func TestAudit_F04_AbortPreservesExistingEmptyFile(t *testing.T) {
	root := t.TempDir()
	put(t, root, ".gitignore", "")
	must(t, stashCapture(root, []string{".gitignore"}))
	put(t, root, ".gitignore", "changed\n")
	must(t, stashRestore(root))
	got, err := os.ReadFile(filepath.Join(root, ".gitignore"))
	if err != nil || len(got) != 0 {
		t.Fatalf("expected existing zero-byte file; exists=%v error=%v", exists(root, ".gitignore"), err)
	}
}
func TestAudit_F05_CorruptStashFailsAndRetainsRecovery(t *testing.T) {
	root := t.TempDir()
	put(t, root, "a.md", "original a\n")
	put(t, root, "b.md", "original b\n")
	must(t, stashCapture(root, []string{"a.md", "b.md"}))
	put(t, root, "a.md", "changed a\n")
	put(t, root, "b.md", "changed b\n")
	key := sha256.Sum256([]byte("a.md"))
	must(t, os.Remove(filepath.Join(root, stashDir, hex.EncodeToString(key[:12])+".bin")))
	err := stashRestore(root)
	a, _ := os.ReadFile(filepath.Join(root, "a.md"))
	if err == nil || !exists(root, stashDir) {
		t.Fatalf("restore must fail and retain recovery: error=%v stashExists=%v a=%q", err, exists(root, stashDir), a)
	}
}
func TestAudit_F06_ReinstallDoesNotInventOriginal(t *testing.T) {
	root := t.TempDir()
	const rel = "docs/WORKFLOW.md"
	// The Install loop invokes captureOriginal before writing a new file.
	must(t, captureOriginal(root, rel))
	put(t, root, rel, "upstream\n")
	// The same captureOriginal call executes on a second Install.
	must(t, captureOriginal(root, rel))
	removeManagedFile(root, rel, []byte("upstream\n"), &strings.Builder{})
	if exists(root, rel) {
		t.Fatal("second capture misclassified generated bytes as a pre-install original; uninstall retained file")
	}
}
func TestAudit_F07_UninstallPreservesPreexistingIgnoreRule(t *testing.T) {
	// appendGitignoreEntries does not append a rule already present.
	// Uninstall nevertheless passes the entire managed list to dropLines.
	before := "# consumer-owned ignore\n/.zharness/\nkeep/\n"
	installed := before + "# zharness v0.15 managed set\n"
	got := dropLines([]byte(installed), []string{"# zharness v0.15 managed set", "/.zharness/"})
	if string(got) != before {
		t.Fatalf("preexisting ignore rule removed: got %q, want %q", got, before)
	}
}
func TestAudit_Control_ExistingAgentsOriginalKeepsSurroundingProse(t *testing.T) {
	root := t.TempDir()
	put(t, root, agentsTarget, "consumer\n")
	must(t, captureOriginal(root, agentsTarget))
	put(t, root, agentsTarget, "consumer\n"+block()+"later\n")
	removeAgentsBlock(root, &strings.Builder{})
	got, err := os.ReadFile(filepath.Join(root, agentsTarget))
	must(t, err)
	if strings.Contains(string(got), blockBegin) || !strings.Contains(string(got), "later") {
		t.Fatalf("unexpected contents: %q", got)
	}
}
func TestAudit_Control_AbortRestoresNonemptyAndAbsent(t *testing.T) {
	root := t.TempDir()
	put(t, root, "existing.md", "original\n")
	must(t, stashCapture(root, []string{"existing.md", "absent.md"}))
	put(t, root, "existing.md", "changed\n")
	put(t, root, "absent.md", "created\n")
	must(t, stashRestore(root))
	got, err := os.ReadFile(filepath.Join(root, "existing.md"))
	must(t, err)
	if string(got) != "original\n" || exists(root, "absent.md") {
		t.Fatal("ordinary restoration failed")
	}
}
func TestAudit_Control_SingleInstallGeneratedFileIsRemoved(t *testing.T) {
	root := t.TempDir()
	const rel = "docs/WORKFLOW.md"
	must(t, captureOriginal(root, rel))
	put(t, root, rel, "upstream\n")
	removeManagedFile(root, rel, []byte("upstream\n"), &strings.Builder{})
	if exists(root, rel) {
		t.Fatal("ordinary generated-file removal failed")
	}
}
